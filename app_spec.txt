<project_specification>
  <project_name>CineGen</project_name>

  <overview>
    CineGen is a SaaS platform for creating long-form AI-generated videos with consistent characters, scenes, and dialogue. Users bring their own API keys for AI services (Hailuo/Kling for video, NanoBanana for images, 11Labs for audio) and use a guided wizard to define their video project. The system automatically generates scene breakdowns, images, video clips, and audio dialogue, stitching everything into cohesive long-form videos.
  </overview>

  <technology_stack>
    <frontend>
      <framework>React</framework>
      <styling>Tailwind CSS</styling>
      <state_management>React Context + hooks</state_management>
      <routing>React Router</routing>
    </frontend>
    <backend>
      <runtime>Node.js with Express</runtime>
      <database>PostgreSQL</database>
      <orm>Prisma</orm>
      <authentication>JWT with email/password</authentication>
    </backend>
    <communication>
      <api>REST API</api>
      <realtime>WebSocket for generation progress</realtime>
    </communication>
    <external_apis>
      <video_generation>Hailuo 2.3 / Kling 2.6 (user's API key)</video_generation>
      <image_generation>NanoBanana Pro / Hailuo (user's API key)</image_generation>
      <audio_generation>11Labs (user's API key)</audio_generation>
      <ai_scene_generation>OpenAI or Anthropic for scene/dialogue generation</ai_scene_generation>
    </external_apis>
  </technology_stack>

    </character_profiles>

      - View list of all projects

    </project_management>

    <project_wizard>
      <step_1_duration>
        - Input total video duration (in seconds or minutes)
        - System calculates number of clips needed (duration / 6 seconds)
        - System picks optimal resolution
      </step_1_duration>
      <step_2_genre>
        - Select genre from list: Action, Drama, Comedy, Horror, Sci-Fi, Romance, Thriller, Documentary
      </step_2_genre>
      <step_3_setting>
        - Text input for time period and location description
      </step_3_setting>
      <step_4_characters>
        - Select from saved character profiles
        - Upload new character inline
        - Assign role/description for each character in this project
        - Up to 6 main characters per project
      </step_4_characters>
      <step_5_plot>
        - Text input for plot/story description
        - User describes the narrative in their own words
      </step_5_plot>
      <step_6_voices>
        - Assign 11Labs voice to each character
        - Voice selection from available 11Labs voices
        - Preview voice sample (if available)
      </step_6_voices>
      <step_7_review>
        - AI generates complete scene breakdown
        - Display scenes with: dialogue, camera angles, emotions, actions
        - Each scene maps to a 6-second clip
        - User can edit scene content
        - User can add new scenes
        - User can delete scenes
        - User can reorder scenes
      </step_7_review>
      <step_8_generate>
        - Start generation button
        - Confirmation before starting
      </step_8_generate>
    </project_wizard>

    <ai_scene_generation>
      - Generate complete scene breakdown from wizard inputs
      - Generate natural dialogue for each character
      - Generate camera angles (wide, medium, close-up, etc.)
      - Generate emotions and actions for characters
      - Break story into 6-second clip segments
      - Maintain narrative consistency across all scenes
      - Support for up to 100+ scenes (for 600+ second videos)
    </ai_scene_generation>

    <image_generation>
      - Generate start image for each clip
      - Generate end image for each clip
      - Maintain character consistency using reference images
      - Maintain scene/setting consistency
      - Use NanoBanana Pro or Hailuo API (user's key)
      - Handle generation errors gracefully
    </image_generation>

    <video_generation>
      - Generate 6-second video clip from start/end image pairs
      - Use Hailuo 2.3 or Kling 2.6 API (user's key)
      - Process clips sequentially or in batches
      - Store generated clips
      - Handle generation errors gracefully
    </video_generation>

    <audio_generation>
      - Generate dialogue audio for each scene
      - Use correct 11Labs voice per character
      - Sync audio timing with video clips
      - Handle generation errors gracefully
    </audio_generation>

    <generation_progress>
      - Real-time progress display on page
      - Show current stage (images, video, audio)
      - Show progress percentage
      - Show current clip being generated
      - WebSocket for live updates
      - Handle errors and display error messages
      - Allow user to stay on page and watch progress
    </generation_progress>

    <clip_review_and_regeneration>
      - View all generated clips in grid/list
      - Play individual clip with audio
      - Regenerate specific clip (image + video + audio)
      - Preview regenerated clip before confirming
      - Replace clip with regenerated version
    </clip_review_and_regeneration>

    <export>
      - Download individual clips as MP4
      - Download combined full video as MP4

      - Stitch all clips with audio into final video
